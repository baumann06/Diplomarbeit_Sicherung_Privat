class AppInstaller:
    """Installer für Desktop-Verknüpfung und App-Konfiguration"""
    
    @staticmethod
    def get_app_directory():
        """Gibt das App-Verzeichnis zurück"""
        if getattr(sys, 'frozen', False):
            # Wenn als exe gestartet
            return os.path.dirname(sys.executable)
        else:
            # Wenn als Python-Script gestartet
            return os.path.dirname(os.path.abspath(__file__))
    
    @staticmethod
    def create_desktop_shortcut():
        """Erstellt Desktop-Verknüpfung"""
        desktop = os.path.join(os.path.expanduser("~"), "Desktop")
        app_dir = AppInstaller.get_app_directory()
        
        if platform.system() == "Windows":
            shortcut_path = os.path.join(desktop, "TimeTracker Pro.lnk")
            target = sys.executable if getattr(sys, 'frozen', False) else f'"{sys.executable}" "{__file__}"'
            
            # Windows Shortcut erstellen
            try:
                import winshell
                winshell.CreateShortcut(
                    Path=shortcut_path,
                    Target=target,
                    Icon=(sys.executable if getattr(sys, 'frozen', False) else None, 0),
                    Description="TimeTracker Pro - Professionelle Zeiterfassung"
                )
                return True
            except ImportError:
                # Alternative ohne winshell
                with open(shortcut_path.replace('.lnk', '.bat'), 'w') as f:
                    f.write(f'@echo off\ncd /d "{app_dir}"\n{target}\npause')
                return True
                
        elif platform.system() == "Darwin":  # macOS
            shortcut_path = os.path.join(desktop, "TimeTracker Pro.command")
            with open(shortcut_path, 'w') as f:
                f.write(f'#!/bin/bash\ncd "{app_dir}"\n{sys.executable} "{__file__}"\n')
            os.chmod(shortcut_path, 0o755)
            return True
            
        else:  # Linux
            shortcut_path = os.path.join(desktop, "TimeTracker Pro.desktop")
            with open(shortcut_path, 'w') as f:
                f.write(f"""[Desktop Entry]
Version=1.0
Type=Application
Name=TimeTracker Pro
Comment=Professionelle Zeiterfassung
Exec={sys.executable} "{__file__}"
Icon=utilities-system-monitor
Terminal=false
Categories=Office;
""")
            os.chmod(shortcut_path, 0o755)
            return True
        
        return False
    
    @staticmethod
    def install_app():
        """Installiert die App (optional)"""
        app_dir = AppInstaller.get_app_directory()
        
        # Prüfen ob bereits installiert
        config_file = os.path.join(app_dir, "timetracker_config.json")
        if os.path.exists(config_file):
            return True
        
        # Konfigurationsdatei erstellen
        config = {
            "version": "2.0",
            "installed": True,
            "install_date": datetime.datetime.now().isoformat(),
            "database_path": os.path.join(app_dir, "timetracker_pro.db")
        }
        
        try:
            with open(config_file, 'w') as f:
                json.dump(config, f, indent=2)
            return True
        except Exception as e:
            print(f"Installation fehlgeschlagen: {e}")
            return False


class TimeTrackerDB:
    """Erweiterte Datenbankschicht für SQLite"""
    
    def __init__(self, db_path=None):
        if db_path is None:
            app_dir = AppInstaller.get_app_directory()
            self.db_path = os.path.join(app_dir, "timetracker_pro.db")
        else:
            self.db_path = db_path
        self.init_database()
    
    def init_database(self):
        """Erstellt die Datenbank-Tabellen mit erweiterten Feldern"""
        conn = sqlite3.connect(self.db_path)
        cursor = conn.cursor()
        
        # Settings-Tabelle für App-Einstellungen
        cursor.execute('''
            CREATE TABLE IF NOT EXISTS settings (
                key TEXT PRIMARY KEY,
                value TEXT NOT NULL,
                created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
                updated_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP
            )
        ''')
        
        # Benutzer-Tabelle erweitert
        cursor.execute('''
            CREATE TABLE IF NOT EXISTS users (
                id INTEGER PRIMARY KEY AUTOINCREMENT,
                name TEXT UNIQUE NOT NULL,
                email TEXT,
                role TEXT DEFAULT 'user',
                active BOOLEAN DEFAULT 1,
                avatar_color TEXT DEFAULT '#2563eb',
                created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP
            )
        ''')
        
        # Projekte-Tabelle (ersetzt Meilensteine)
        cursor.execute('''
            CREATE TABLE IF NOT EXISTS projects (
                id INTEGER PRIMARY KEY AUTOINCREMENT,
                name TEXT UNIQUE NOT NULL,
                description TEXT,
                color TEXT DEFAULT '#2563eb',
                client TEXT,
                active BOOLEAN DEFAULT 1,
                created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP
            )
        ''')
        
        # Zeiteinträge-Tabelle erweitert
        cursor.execute('''
            CREATE TABLE IF NOT EXISTS time_entries (
                id INTEGER PRIMARY KEY AUTOINCREMENT,
                start_time TIMESTAMP NOT NULL,
                end_time TIMESTAMP NOT NULL,
                duration INTEGER NOT NULL,
                project_id INTEGER,
                description TEXT,
                user_id INTEGER,
                tags TEXT,
                hourly_rate REAL DEFAULT 0,
                billable BOOLEAN DEFAULT 0,
                status TEXT DEFAULT 'completed',
                created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
                updated_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
                FOREIGN KEY (project_id) REFERENCES projects (id),
                FOREIGN KEY (user_id) REFERENCES users (id)
            )
        ''')
        
        # Kategorien für bessere Organisation
        cursor.execute('''
            CREATE TABLE IF NOT EXISTS categories (
                id INTEGER PRIMARY KEY AUTOINCREMENT,
                name TEXT UNIQUE NOT NULL,
                color TEXT DEFAULT '#64748b',
                icon TEXT DEFAULT '📋',
                active BOOLEAN DEFAULT 1,
                created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP
            )
        ''')
        
        # Projekt-Kategorien Verknüpfung
        cursor.execute('''
            CREATE TABLE IF NOT EXISTS project_categories (
                project_id INTEGER,
                category_id INTEGER,
                PRIMARY KEY (project_id, category_id),
                FOREIGN KEY (project_id) REFERENCES projects (id),
                FOREIGN KEY (category_id) REFERENCES categories (id)
            )
        ''')
        
        conn.commit()
        conn.close()
        
        # Standarddaten einfügen
        self.insert_default_data()
    
    def insert_default_data(self):
        """Fügt Standarddaten ein"""
        # Standard-Benutzer
        if not self.get_users():
            self.add_user("Standard Benutzer", "user@timetracker.local", avatar_color="#2563eb")
        
        # Standard-Projekt
        if not self.get_projects():
            self.add_project("Allgemein", "Standard-Projekt für allgemeine Aufgaben", "#10b981")
            self.add_project("Administration", "Administrative Tätigkeiten", "#f59e0b")
            self.add_project("Entwicklung", "Software-Entwicklung", "#3b82f6")
        
        # Standard-Kategorien
        if not self.get_categories():
            categories = [
                ("Büroarbeit", "#64748b", "💼"),
                ("Entwicklung", "#3b82f6", "💻"),
                ("Meeting", "#f59e0b", "🤝"),
                ("Lernen", "#10b981", "📚"),
                ("Support", "#ef4444", "🛠️")
            ]
            for name, color, icon in categories:
                self.add_category(name, color, icon)
    
    def add_user(self, name, email="", role="user", avatar_color="#2563eb"):
        """Fügt einen neuen Benutzer hinzu"""
        conn = sqlite3.connect(self.db_path)
        cursor = conn.cursor()
        try:
            cursor.execute("INSERT INTO users (name, email, role, avatar_color) VALUES (?, ?, ?, ?)", 
                         (name, email, role, avatar_color))
            conn.commit()
            return cursor.lastrowid
        except sqlite3.IntegrityError:
            return None
        finally:
            conn.close()
    
    def get_users(self):
        """Gibt alle aktiven Benutzer zurück"""
        conn = sqlite3.connect(self.db_path)
        cursor = conn.cursor()
        cursor.execute("SELECT id, name, email, avatar_color FROM users WHERE active = 1 ORDER BY name")
        result = cursor.fetchall()
        conn.close()
        return result
    
    def add_project(self, name, description="", color="#2563eb", client=""):
        """Fügt ein neues Projekt hinzu"""
        conn = sqlite3.connect(self.db_path)
        cursor = conn.cursor()
        try:
            cursor.execute("INSERT INTO projects (name, description, color, client) VALUES (?, ?, ?, ?)", 
                         (name, description, color, client))
            conn.commit()
            return cursor.lastrowid
        except sqlite3.IntegrityError:
            return None
        finally:
            conn.close()
    
    def get_projects(self):
        """Gibt alle aktiven Projekte zurück"""
        conn = sqlite3.connect(self.db_path)
        cursor = conn.cursor()
        cursor.execute("SELECT id, name, description, color, client FROM projects WHERE active = 1 ORDER BY name")
        result = cursor.fetchall()
        conn.close()
        return result
    
    def add_category(self, name, color="#64748b", icon="📋"):
        """Fügt eine neue Kategorie hinzu"""
        conn = sqlite3.connect(self.db_path)
        cursor = conn.cursor()
        try:
            cursor.execute("INSERT INTO categories (name, color, icon) VALUES (?, ?, ?)", 
                         (name, color, icon))
            conn.commit()
            return cursor.lastrowid
        except sqlite3.IntegrityError:
            return None
        finally:
            conn.close()
    
    def get_categories(self):
        """Gibt alle aktiven Kategorien zurück"""
        conn = sqlite3.connect(self.db_path)
        cursor = conn.cursor()
        cursor.execute("SELECT id, name, color, icon FROM categories WHERE active = 1 ORDER BY name")
        result = cursor.fetchall()
        conn.close()
        return result
    
    def add_time_entry(self, start_time, end_time, duration, project_id, description, user_id, tags="", hourly_rate=0, billable=False):
        """Fügt einen Zeiteintrag hinzu"""
        conn = sqlite3.connect(self.db_path)
        cursor = conn.cursor()
        cursor.execute('''
            INSERT INTO time_entries (start_time, end_time, duration, project_id, description, user_id, tags, hourly_rate, billable)
            VALUES (?, ?, ?, ?, ?, ?, ?, ?, ?)
        ''', (start_time, end_time, duration, project_id, description, user_id, tags, hourly_rate, billable))
        conn.commit()
        entry_id = cursor.lastrowid
        conn.close()
        return entry_id
    
    def get_time_entries(self, limit=100, filter_date=None):
        """Gibt Zeiteinträge mit Projekt- und Benutzernamen zurück"""
        conn = sqlite3.connect(self.db_path)
        cursor = conn.cursor()
        
        query = '''
            SELECT te.id, te.start_time, te.end_time, te.duration,
                   p.name as project, te.description, u.name as user,
                   te.tags, te.hourly_rate, te.billable, p.color, te.status
            FROM time_entries te
            LEFT JOIN projects p ON te.project_id = p.id
            LEFT JOIN users u ON te.user_id = u.id
        '''
        
        params = []
        if filter_date:
            query += " WHERE DATE(te.start_time) = ?"
            params.append(filter_date)
        
        query += " ORDER BY te.start_time DESC LIMIT ?"
        params.append(limit)
        
        cursor.execute(query, params)
        result = cursor.fetchall()
        conn.close()
        return result
    
    def update_time_entry(self, entry_id, start_time, end_time, duration, project_id, description, user_id, tags="", hourly_rate=0, billable=False):
        """Aktualisiert einen Zeiteintrag"""
        conn = sqlite3.connect(self.db_path)
        cursor = conn.cursor()
        cursor.execute('''
            UPDATE time_entries 
            SET start_time=?, end_time=?, duration=?, project_id=?, description=?, user_id=?, 
                tags=?, hourly_rate=?, billable=?, updated_at=CURRENT_TIMESTAMP
            WHERE id=?
        ''', (start_time, end_time, duration, project_id, description, user_id, tags, hourly_rate, billable, entry_id))
        conn.commit()
        conn.close()
    
    def delete_time_entry(self, entry_id):
        """Löscht einen Zeiteintrag"""
        conn = sqlite3.connect(self.db_path)
        cursor = conn.cursor()
        cursor.execute("DELETE FROM time_entries WHERE id=?", (entry_id,))
        conn.commit()
        conn.close()
    
    def get_statistics(self):
        """Gibt erweiterte Statistiken zurück"""
        conn = sqlite3.connect(self.db_path)
        cursor = conn.cursor()
        
        # Heute
        today = datetime.date.today()
        cursor.execute('''
            SELECT COALESCE(SUM(duration), 0) 
            FROM time_entries 
            WHERE DATE(start_time) = ?
        ''', (today,))
        today_seconds = cursor.fetchone()[0]
        
        # Diese Woche
        week_start = today - datetime.timedelta(days=today.weekday())
        cursor.execute('''
            SELECT COALESCE(SUM(duration), 0) 
            FROM time_entries 
            WHERE DATE(start_time) >= ?
        ''', (week_start,))
        week_seconds = cursor.fetchone()[0]
        
        # Diesen Monat
        month_start = today.replace(day=1)
        cursor.execute('''
            SELECT COALESCE(SUM(duration), 0) 
            FROM time_entries 
            WHERE DATE(start_time) >= ?
        ''', (month_start,))
        month_seconds = cursor.fetchone()[0]
        
        # Gesamtzahl Einträge
        cursor.execute('SELECT COUNT(*) FROM time_entries')
        total_entries = cursor.fetchone()[0]
        
        # Durchschnittliche tägliche Arbeitszeit (letzte 30 Tage)
        cursor.execute('''
            SELECT AVG(daily_duration) FROM (
                SELECT SUM(duration) as daily_duration
                FROM time_entries 
                WHERE DATE(start_time) >= DATE('now', '-30 days')
                GROUP BY DATE(start_time)
            )
        ''')
        avg_daily = cursor.fetchone()[0] or 0
        
        # Aktive Projekte
        cursor.execute('SELECT COUNT(*) FROM projects WHERE active = 1')
        active_projects = cursor.fetchone()[0]
        
        conn.close()
        
        return {
            'today': today_seconds,
            'week': week_seconds,
            'month': month_seconds,
            'total_entries': total_entries,
            'avg_daily': avg_daily,
            'active_projects': active_projects
        }
    
    def get_project_statistics(self, days=30):
        """Gibt Projekt-spezifische Statistiken zurück"""
        conn = sqlite3.connect(self.db_path)
        cursor = conn.cursor()
        
        cursor.execute('''
            SELECT p.name, p.color, SUM(te.duration) as total_duration, COUNT(te.id) as entry_count
            FROM time_entries te
            JOIN projects p ON te.project_id = p.id
            WHERE DATE(te.start_time) >= DATE('now', '-{} days')
            GROUP BY p.id, p.name, p.color
            ORDER BY total_duration DESC
        '''.format(days))
        
        result = cursor.fetchall()
        conn.close()
        return result
